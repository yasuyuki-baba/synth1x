cmake_minimum_required(VERSION 3.25)
include(cmake/Synth1XVersion.cmake)
set(SYNTH1X_RELEASE_TAG "" CACHE STRING "Release tag for source archive/manual release builds; empty means a dated development build")
synth1x_configure_version("${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated/version")
project(Synth1X VERSION ${SYNTH1X_NUMERIC_VERSION} LANGUAGES C CXX)
# Out-of-source layout based on iPlug2/iPlug2OOS (see dependencies.lock.json).
set(IPLUG2_DIR "${CMAKE_CURRENT_SOURCE_DIR}/iPlug2" CACHE PATH "Path to the iPlug2 framework checkout")
# Experimental Linux native builds are selected through the linux presets.
if(WIN32 OR APPLE)
    set(synth1x_plugin_default ON)
else()
    set(synth1x_plugin_default OFF)
endif()
option(SYNTH1X_BUILD_IPLUG2 "Build the iPlug2 desktop instrument" ${synth1x_plugin_default})
option(SYNTH1X_BUILD_TESTS "Build portable core regression tests" ON)
option(SYNTH1X_BUILD_NATIVE_SMOKE "Test built plugins through their public state/MIDI/audio APIs" OFF)
option(SYNTH1X_BUILD_TOOLS "Build offline development tools" ON)
if(SYNTH1X_BUILD_IPLUG2 AND MSVC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(cmake/Synth1XCore.cmake)
if(SYNTH1X_BUILD_TOOLS OR SYNTH1X_BUILD_TESTS)
    include(cmake/Synth1XTools.cmake)
endif()
if(SYNTH1X_BUILD_IPLUG2)
    add_subdirectory(projects/Synth1X)
endif()
if(SYNTH1X_BUILD_TESTS OR SYNTH1X_BUILD_NATIVE_SMOKE)
    include(CTest)
endif()
if(SYNTH1X_BUILD_TESTS)
    include(cmake/PortableTests.cmake)
endif()
if(SYNTH1X_BUILD_NATIVE_SMOKE)
    include(cmake/NativeSmokeTests.cmake)
endif()
